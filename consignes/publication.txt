# ═══════════════════════════════════════════════════════════════════════════
# LE RYTHME DE PUBLICATION
# ═══════════════════════════════════════════════════════════════════════════
#
# Les anecdotes sont écrites d'avance et gardées en réserve. Ce fichier dit
# quand elles sortent, et combien. Vous pouvez le modifier ici sur GitHub, ou
# depuis console.html — onglet Publication — qui l'écrit pour vous.
#
# L'action « 5 · Publier » tourne TOUS LES JOURS à 6 h UTC. C'est ce fichier,
# et lui seul, qui décide si aujourd'hui est un jour de publication.
#
# ─────────────────────────────────────────────────────────────────────────
# À QUELLE FRÉQUENCE
#   quotidien      tous les jours                                (défaut)
#   hebdomadaire   une fois par semaine, le jour indiqué par « jours »
#   jours          seulement les jours listés dans « jours »
#   pause          rien ne sort, la réserve dort
# ─────────────────────────────────────────────────────────────────────────
rythme: quotidien

# Les jours, pour « hebdomadaire » et « jours ».
# 1 = lundi, 2 = mardi, 3 = mercredi, 4 = jeudi, 5 = vendredi,
# 6 = samedi, 7 = dimanche.   Exemple :  jours: 2,5
jours: 1

# ─────────────────────────────────────────────────────────────────────────
# COMBIEN de sujets à chaque passage
# UN SUJET, PAS UNE FICHE : publier « Lac Nyos » met en ligne le français ET
# l'anglais le même jour — si les deux sont publiés, voir « langues ».
# ─────────────────────────────────────────────────────────────────────────
parPassage: 3

# ─────────────────────────────────────────────────────────────────────────
# OU BIEN : TOUT SORTIR D'ICI UNE DATE
# Si cette ligne porte une date, « parPassage » est ignoré : la réserve est
# répartie sur les passages qui restent d'ici là, et le nombre se recalcule
# à chaque exécution. Laissez vide pour ne pas l'utiliser.
#     jusqu-au: 2026-12-31
# ─────────────────────────────────────────────────────────────────────────
jusqu-au:

# ─────────────────────────────────────────────────────────────────────────
# QUELLES LANGUES SONT PUBLIÉES
#   fr        le français seul — l'anglais reste au dépôt, non publié,
#             et le bouton FR/EN disparaît du site et de l'application
#   fr,en     les deux                                            (défaut)
#
# Écrire l'anglais sans le publier est un état parfaitement valable : le jour
# où vous ajoutez « en » ici, tout ce qui est écrit et contrôlé sort.
#
# Ce réglage agit AVANT toute publication : les outils le recopient dans
# catalog.json et dans anecdotes/index.json, que le site et l'application
# lisent au démarrage. Pour l'appliquer sans attendre une moisson :
# Actions -> Entretien -> recompter (quelques secondes, gratuit).
# ─────────────────────────────────────────────────────────────────────────
langues: fr

# ─────────────────────────────────────────────────────────────────────────
# LES PHOTOS DES FICHES
#   oui       la photo de l'article, sous un voile appuyé — le texte prime
#             et l'image reste là, comme une ambiance                (défaut)
#   franches  la photo à nu, sans voile ajouté : l'image prime
#   non       aucune photo. Chaque fiche garde son art dessiné, composé pour
#             être lu par-dessus. Rien n'est perdu : les photos restent
#             enregistrées, remettre « oui » les fait revenir.
#
# Ce réglage est gravé dans les pages à la construction : il s'applique dès
# la première seconde, hors ligne compris. Il ne coûte rien et ne réécrit
# aucune fiche.
# ─────────────────────────────────────────────────────────────────────────
images: oui

# ─────────────────────────────────────────────────────────────────────────
# DANS QUEL ORDRE
#   potentiel   les sujets les plus prometteurs d'abord           (défaut)
#   note        les mieux notés par le rédacteur d'abord — garde le meilleur
#               pour tout de suite, quitte à s'appauvrir avec le temps
#   hasard      au hasard — la réserve se vide uniformément
# ─────────────────────────────────────────────────────────────────────────
ordre: potentiel

# ─────────────────────────────────────────────────────────────────────────
# PUBLIER MÊME CE QUI EST EN QUARANTAINE ? Non, évidemment. Ne mettez « oui »
# que si vous savez exactement pourquoi.
# ─────────────────────────────────────────────────────────────────────────
quarantaine: non

# Actions → « 5 · Publier » → Run workflow, en indiquant un nombre. Une
# publication lancée à la main passe outre le rythme : c'est vous qui décidez.
