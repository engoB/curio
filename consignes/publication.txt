#
# rythme : quotidien | hebdomadaire | jours | pause

# 1 = lundi … 7 = dimanche

# Combien de sujets à chaque passage
parPassage: 7

# Ou : tout sortir d’ici cette date — « parPassage » est alors ignoré
jusqu-au: 

# Langues publiées : fr | en | fr,en

# Photos des fiches : oui (voilées) | franches (à nu) | non

# Le passage de la relecture au STOCK prêt à publier.
#   manuel  vous appuyez sur « Envoyer au stock » dans 3 · Relecture

# Dans les deux cas, valider ne met JAMAIS en ligne : c’est le rythme
# ci-dessus qui décide quand une fiche du stock sort.

# Ordre : potentiel | note | hasard
ordre: potentiel

# Publier ce qui est en quarantaine ? Non.
